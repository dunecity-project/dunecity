cmake_minimum_required(VERSION 3.15)
project(DesktopUpdaterTests LANGUAGES CXX)
enable_testing()
set(CMAKE_CXX_STANDARD 17)
find_package(OpenSSL REQUIRED)
get_filename_component(DUNE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
add_executable(desktop_updater_tests test.cpp ${DUNE_ROOT}/src/Network/UpdateManifest.cpp)
target_include_directories(desktop_updater_tests PRIVATE ${DUNE_ROOT}/include)
target_link_libraries(desktop_updater_tests PRIVATE OpenSSL::Crypto)
if(UNIX)
    target_sources(desktop_updater_tests PRIVATE ${DUNE_ROOT}/src/misc/AppImageUpdate.cpp)
endif()
if(COMMAND dunecity_add_test)
    dunecity_add_test(headless integration NAME desktop_updater_security COMMAND desktop_updater_tests)
else()
    add_test(NAME desktop_updater_security COMMAND desktop_updater_tests)
    set_tests_properties(desktop_updater_security PROPERTIES LABELS "headless;integration;nightly")
endif()
