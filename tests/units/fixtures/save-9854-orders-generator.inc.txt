// Scratch generator: writes a SAVEGAMEVERSION 9854 save with live unit orders and
// a producing factory, using the unmodified base build (40f4e7a8).
#include <Game.h>
#include <House.h>
#include <Map.h>
#include <sand.h>
#include <mod/ModManager.h>
#include <units/UnitBase.h>
#include <structures/BuilderBase.h>
#include <fstream>
#include <memory>

int runUnitSpeedProbe() {
    auto require = [](bool ok, const char* why) { if(!ok) throw std::runtime_error(why); };
    require(ModManager::instance().setActiveMod("vanilla"), "mod");
    settings.general.playerName = "Legacy";
    std::ifstream map(PROBE_MAP_PATH);
    const std::string mapData((std::istreambuf_iterator<char>(map)), {});
    auto game = std::make_unique<Game>();
    currentGame = game.get();
    GameInitSettings init("legacy.ini", mapData, "Legacy 9854", true, settings.gameOptions);
    init.randomSeed = 9854;
    GameInitSettings::HouseInfo humans(HOUSE_ATREIDES, 1);
    humans.addPlayerInfo({"Legacy", "HumanPlayer"});
    init.addHouseInfo(humans);
    GameInitSettings::HouseInfo enemies(HOUSE_HARKONNEN, 2);
    enemies.addPlayerInfo({"Enemy", "HumanPlayer"});
    init.addHouseInfo(enemies);
    game->initGame(init);
    game->initializeGameLoop();
    auto* owner = game->getHouse(HOUSE_ATREIDES);
    auto* enemy = game->getHouse(HOUSE_HARKONNEN);
    require(owner && enemy, "houses");
    const Coord site(20,12);
    for(int y=site.y-3;y<=site.y+8;++y) for(int x=site.x-3;x<=site.x+14;++x) {
        auto* tile=currentGameMap->getTile(x,y); tile->setType(Terrain_Rock); tile->setRoad(false);
    }
    auto* factory = dynamic_cast<BuilderBase*>(owner->placeStructure(NONE_ID, Structure_HeavyFactory, site.x, site.y+5, true, true));
    require(factory, "factory");
    factory->updateBuildList();
    owner->addCredits(5000, true);
    factory->doProduceItem(Unit_Tank, false);
    factory->doProduceItem(Unit_Tank, false);
    std::vector<UnitBase*> movers;
    for(int i=0;i<3;++i) {
        auto* tank = owner->createUnit(Unit_Tank, true); tank->deploy(site+Coord(i,0));
        tank->doMove2Pos(site.x+10, site.y+i, true);
        movers.push_back(tank);
    }
    auto* guard = owner->createUnit(Unit_Launcher, true); guard->deploy(site+Coord(0,2));
    guard->doSetAttackMode(AREAGUARD);
    auto* foe = enemy->createUnit(Unit_Tank, true); foe->deploy(site+Coord(13,4));
    foe->doSetAttackMode(GUARD);
    for(int i=0;i<90;++i) game->updateGameState();
    const std::string out = std::getenv("UNIT_SPEED_PROBE_OUT");
    const std::string path = out + "/save-9854-orders.dls";
    require(game->saveGame(path), "save");
    SDL_Log("LEGACY_9854_SAVE: version=%d units=%zu factory=%u movers=%u,%u,%u launcher=%u foe=%u",
        SAVEGAMEVERSION, unitList.size(), factory->getObjectID(), movers[0]->getObjectID(),
        movers[1]->getObjectID(), movers[2]->getObjectID(), guard->getObjectID(), foe->getObjectID());
    SDL_Log("UNIT_SPEED_PROBE_PASS: legacy save written");
    return -1;
}
